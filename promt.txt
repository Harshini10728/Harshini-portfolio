<script>
    function setProgress(circleId, percent) {
  const circle = document.querySelector(`#${circleId} .progress-circle__bar`);
  const text = document.querySelector(`#${circleId} .progress-circle__text`);

  const radius = circle.r.baseVal.value;
  const circumference = 2 * Math.PI * radius;

  const offset = circumference - percent / 100 * circumference;
  circle.style.strokeDasharray = `${circumference} ${circumference}`;
  circle.style.strokeDashoffset = offset;

  text.textContent = `${percent}%`;
}

// Example usage:
setProgress('circle1', 90); // Set progress of circle 1 to 70%
setProgress('circle2', 70); // Set progress of circle 2 to 50%
setProgress('circle3', 75); // Set progress of circle 3 to 85%
</script>
